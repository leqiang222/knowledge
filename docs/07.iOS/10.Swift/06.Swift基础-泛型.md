### 泛型函数

泛型函数可适用于任意类型，下面是参数交换函数的泛型版本

```swift
func swapTwoValue<T>(_ value1: inout T, _ value2: inout T) {
    let tmp = value1
    value1 = value2
    value2 = tmp
}

var a = "a"
var b = "b"
var c = 1
var d = 2

swapTwoValue(&a, &b)
swapTwoValue(&c, &d)
```

类型约束：在一个类型参数名后面放置一个类名或者协议名，并用冒号进行分隔，来定义类型约束。

```swift
func someFunction<T: SomeClass, U: SomeProtocol>(someT: T, someU: U) {
    // 这里是泛型函数的函数体部分
}

func findIndex<T>(of valueToFind: T, in array:[T]) -> Int? {
    for (index, value) in array.enumerated() {
        if value == valueToFind {
            return index
        }
    }
    return nil
}
```
